ORB Pseudocode (with spatial calibration)

Input:
  R         reference image
  T         defective image
Output:
  L_boxes   predicted defect bounding boxes
  H         homography matrix (or None if calibration failed/disabled)

Begin

--- Stage 0: Preprocessing ---
  (R_gray, T_gray) = preprocess_pair(R, T, preprocessing_config)

--- Stage 1: Spatial Calibration (when calibrate = true) ---
  1:  detector <- ORB_Create(nfeatures, scaleFactor, nlevels)
  2:  (K_R1, D_R1) <- detector.DetectAndCompute(R_gray)
  3:  (K_T1, D_T1) <- detector.DetectAndCompute(T_gray)
  4:  matcher1 <- BFMatcher(normType = NORM_HAMMING, crossCheck = false)
  5:  knn_matches <- matcher1.KnnMatch(D_T1, D_R1, k = 2)
  6:  S_src <- {}, S_dst <- {}
  7:  for each pair (m, n) in knn_matches:
  8:      if LoweRatioTest(m, n, r) passes:
  9:          S_src.add(K_T1[m.queryIdx].pt)
 10:          S_dst.add(K_R1[m.trainIdx].pt)
 11:      end if
 12:  end for
 13:  (H, inlier_mask) <- FindHomography(S_src, S_dst, RANSAC, rr)
 14:  if H exists and countInliers(inlier_mask) >= 4:
 15:      T_aligned <- WarpPerspective(T_gray, H, sizeOf(R_gray))
 16:      T_gray <- T_aligned
 17:  else:
 18:      H <- None
 19:  end if

--- Stage 2: Defect Detection via Feature Anomaly ---
 20:  (K_R2, D_R2) <- Detector.DetectAndCompute(R_gray)
 21:  (K_A2, D_A2) <- Detector.DetectAndCompute(T_gray)
 22:  matcher2 <- BFMatcher(normType = NORM_HAMMING, crossCheck = true)
 23:  matches <- matcher2.Match(D_R2, D_A2)
 24:  matched_def_indices <- {}
 25:  defect_points <- {}
 26:  for each match m in matches:
 27:      matched_def_indices.add(m.trainIdx)
 28:      pt_ref <- K_R2[m.queryIdx].pt
 29:      pt_def <- K_A2[m.trainIdx].pt
 30:      disp <- EuclideanDistance(pt_def, pt_ref)
 31:      if m.distance > h_thresh or disp > d_thresh:
 32:          defect_points.add(pt_def)    [inconsistent match]
 33:      end if
 34:  end for
 35:  for each keypoint kp at index idx in K_A2:
 36:      if idx not in matched_def_indices:
 37:          defect_points.add(kp.pt)    [unmatched keypoint]
 38:      end if
 39:  end for
 40:  L_boxes <- {}
 41:  for each point (x, y) in defect_points:
 42:      box <- [max(0, x-br), max(0, y-br), min(W, x+br), min(Height0, y+br)]
 43:      L_boxes.add(box)
 44:  end for

  return L_boxes, H

End
